// ============================================================================
//  ESP32 Autonomous Drone Mission Manager
// ============================================================================
//
//  Bridges a Pixhawk (UART2, MAVLink) to a laptop over WiFi:
//    - UDP 14550: raw MAVLink passthrough (QGroundControl / Mission Planner)
//    - UDP 14551: JSON mission API (see Missions/*.py)
//
//  A stored mission is a JSON array of steps, executed one at a time. Two
//  mission styles are both supported and can be freely mixed in one mission:
//
//    1. Pure Guided-mode steps: "reposition" / "fly_to" (absolute lat/lon
//       target), "local_ned" (position offset from wherever the vehicle is
//       when the step starts) and "brake_hover" (explicit zero-velocity
//       hold) all keep the vehicle in GUIDED and drive it directly via
//       MAVLink position/velocity targets.
//    2. Mode-switching steps: "set_mode" hands control to one of ArduPilot's
//       own flight modes (e.g. LOITER) using its own tuned position-hold
//       controller instead of this firmware's guided setpoints.
//
//  "relay_on"/"relay_off" toggle GPIO4, wired only to the ESP32 (not the
//  Pixhawk) for a sprayer/pump relay. No mission below uses the relay yet —
//  it's here so a future spray mission can add relay_on/relay_off steps.
//
// ============================================================================

// ─── MAVLink Parser ──────────────────────────────────────────────────────────
#define MAV_MAX_PAYLOAD 255
static uint8_t  mp_buf[10 + MAV_MAX_PAYLOAD + 2 + 13];
static uint8_t  mp_state       = 0;
static uint8_t  mp_magic       = 0;
static uint8_t  mp_payload_len = 0;
static uint8_t  mp_header_len  = 0;
static uint8_t  mp_signature_len = 0;
static uint8_t  mp_sig_idx     = 0;
static uint32_t mp_msg_id      = 0;
static uint16_t mp_idx         = 0;
static uint8_t  mp_seq         = 0;

static uint16_t crc_acc(uint8_t b, uint16_t crc) {
  uint8_t t = b ^ (uint8_t)(crc & 0xFF);
  t ^= (t << 4);
  return (crc >> 8) ^ ((uint16_t)t << 8) ^ ((uint16_t)t << 3) ^ ((uint16_t)t >> 4);
}

static uint8_t crc_extra(uint32_t id) {
  switch (id) {
    case MAVLINK_MSG_HEARTBEAT:           return 50;
    case MAVLINK_MSG_GPS_RAW_INT:         return 24;
    case MAVLINK_MSG_GLOBAL_POSITION_INT: return 104;
    case MAVLINK_MSG_VFR_HUD:             return 20;
    case MAVLINK_MSG_COMMAND_ACK:         return 143;
    case MAVLINK_MSG_EKF_STATUS_REPORT:   return 71;
    case MAVLINK_MSG_EXTENDED_SYS_STATE:  return 130;
    default: return 0;
  }
}

static void mavlink_dispatch(uint32_t msg_id, uint8_t payload_len, uint8_t* payload) {
  if (msg_id == MAVLINK_MSG_HEARTBEAT && payload_len >= 9) {
    uint32_t cm; memcpy(&cm, payload, 4);
    drone_custom_mode = cm;
    drone_base_mode   = payload[6];
    drone_armed       = (payload[6] & 0x80) != 0;
    last_heartbeat_ms = millis();
    Serial.printf("[HB] armed=%d base_mode=0x%02X custom_mode=%u\n",
                  (int)drone_armed, drone_base_mode, drone_custom_mode);
  }
  else if (msg_id == MAVLINK_MSG_GLOBAL_POSITION_INT && payload_len >= 28) {
    int32_t lat_r, lon_r, ralt_r;
    memcpy(&lat_r,  payload + 4,  4);
    memcpy(&lon_r,  payload + 8,  4);
    memcpy(&ralt_r, payload + 16, 4);
    current_lat = (float)lat_r  / 1e7f;
    current_lon = (float)lon_r  / 1e7f;
    current_alt = (float)ralt_r / 1000.0f;
    last_gps_ms = millis();
    gps_valid = (gps_fix_type >= 3);
    Serial.printf("[GPS] lat=%.6f lon=%.6f alt=%.2fm\n",
                  current_lat, current_lon, current_alt);
  }
  else if (msg_id == MAVLINK_MSG_GPS_RAW_INT && payload_len >= 30) {
    gps_fix_type = payload[28];
    gps_satellites = payload[29];
    last_gps_raw_ms = millis();
    gps_valid = (gps_fix_type >= 3 && last_gps_ms != 0);
    Serial.printf("[GPS_RAW] fix=%u sats=%u\n", gps_fix_type, gps_satellites);
  }
  else if (msg_id == MAVLINK_MSG_COMMAND_ACK && payload_len >= 3) {
    memcpy((void*)&last_ack_command, payload, 2);
    last_ack_result = payload[2];
    last_ack_ms = millis();
    Serial.printf("[ACK] command=%u result=%u\n", last_ack_command, last_ack_result);
  }
  else if (msg_id == MAVLINK_MSG_EKF_STATUS_REPORT && payload_len >= 2) {
    memcpy((void*)&ekf_flags, payload, 2);
    last_ekf_ms = millis();
    Serial.printf("[EKF] flags=0x%04X\n", ekf_flags);
  }
  else if (msg_id == MAVLINK_MSG_EXTENDED_SYS_STATE && payload_len >= 2) {
    drone_landed_state = payload[1];
    Serial.printf("[EXT_STATE] landed_state=%u\n", drone_landed_state);
  }
}

static void mavlink_verify_and_dispatch() {
  uint16_t crc = 0xFFFF;
  int crc_len = mp_header_len + mp_payload_len;
  for (int i = 1; i < crc_len; i++) crc = crc_acc(mp_buf[i], crc);
  crc = crc_acc(crc_extra(mp_msg_id), crc);

  int crc_offset = mp_header_len + mp_payload_len;
  uint16_t recv = (uint16_t)mp_buf[crc_offset] | ((uint16_t)mp_buf[crc_offset + 1] << 8);
  if (crc == recv) mavlink_dispatch(mp_msg_id, mp_payload_len, &mp_buf[mp_header_len]);
}

static void mavlink_parse_byte(uint8_t b) {
  switch (mp_state) {
    case 0:
      if (b == 0xFE || b == 0xFD) {
        mp_magic = b;
        mp_buf[0] = b;
        mp_state = 1;
      }
      break;

    case 1:
      mp_payload_len = b;
      mp_buf[1] = b;
      mp_idx = 0;
      mp_msg_id = 0;
      mp_signature_len = 0;
      mp_sig_idx = 0;
      mp_state = 2;
      break;

    case 2:
      mp_buf[2] = b;
      if (mp_magic == 0xFE) {
        mp_seq = b;
        mp_state = 3;
      } else {
        mp_signature_len = (b & 0x01) ? 13 : 0;
        mp_state = 13;
      }
      break;

    case 3: mp_buf[3]=b; mp_state=4; break;
    case 4: mp_buf[4]=b; mp_state=5; break;
    case 5:
      mp_buf[5]=b;
      mp_msg_id = b;
      mp_header_len = 6;
      mp_state = (mp_payload_len > 0) ? 30 : 31;
      break;

    case 13: mp_buf[3]=b; mp_state=14; break;  // MAVLink2 compat flags
    case 14: mp_seq=b; mp_buf[4]=b; mp_state=15; break;
    case 15: mp_buf[5]=b; mp_state=16; break;
    case 16: mp_buf[6]=b; mp_state=17; break;
    case 17: mp_buf[7]=b; mp_state=18; break;
    case 18: mp_buf[8]=b; mp_state=19; break;
    case 19:
      mp_buf[9]=b;
      mp_msg_id = (uint32_t)mp_buf[7] | ((uint32_t)mp_buf[8] << 8) | ((uint32_t)mp_buf[9] << 16);
      mp_header_len = 10;
      mp_state = (mp_payload_len > 0) ? 30 : 31;
      break;

    case 30:
      mp_buf[mp_header_len + mp_idx++] = b;
      if (mp_idx >= mp_payload_len) mp_state = 31;
      break;

    case 31:
      mp_buf[mp_header_len + mp_payload_len] = b;
      mp_state = 32;
      break;

    case 32:
      mp_buf[mp_header_len + mp_payload_len + 1] = b;
      mavlink_verify_and_dispatch();
      mp_state = (mp_signature_len > 0) ? 33 : 0;
      break;

    case 33:
      mp_buf[mp_header_len + mp_payload_len + 2 + mp_sig_idx++] = b;
      if (mp_sig_idx >= mp_signature_len) mp_state = 0;
      break;

    default: mp_state = 0;
  }
}

// ─── UDP Bridges ─────────────────────────────────────────────────────────────
void forwardMavlinkToUdp(const uint8_t* data, size_t len) {
  if (len == 0) return;
  mavUdp.beginPacket(mavTargetIP, MAVLINK_UDP_PORT);
  mavUdp.write(data, len);
  mavUdp.endPacket();
}

void parsePixhawkTelemetry() {
  while (Serial2.available()) {
    uint8_t buf[128];
    size_t n = 0;
    while (Serial2.available() && n < sizeof(buf)) {
      uint8_t b = (uint8_t)Serial2.read();
      mavlink_parse_byte(b);
      buf[n++] = b;
    }
    forwardMavlinkToUdp(buf, n);
  }
}

void handleMavlinkUdp() {
  int packetSize = mavUdp.parsePacket();
  if (!packetSize) return;

  mavTargetIP = mavUdp.remoteIP();

  uint8_t buf[256];
  while (packetSize > 0) {
    int toRead = packetSize < (int)sizeof(buf) ? packetSize : (int)sizeof(buf);
    int len = mavUdp.read(buf, toRead);
    if (len <= 0) break;
    Serial2.write(buf, len);
    packetSize -= len;
  }
}

void sendUdpText(const char* text) {
  missionUdp.beginPacket(missionUdp.remoteIP(), missionUdp.remotePort());
  missionUdp.write((const uint8_t*)text, strlen(text));
  missionUdp.endPacket();
}

void sendStatusReply() {
  JsonDocument status;
  JsonArray steps = missionDoc["steps"];
  int totalSteps = steps ? (int)steps.size() : 0;
  const char* action = "";
  if (steps && currentStep < totalSteps) {
    action = steps[currentStep]["action"] | "";
  }

  status["type"] = "status";
  status["mission_stored"] = steps ? true : false;
  status["running"] = missionRunning;
  status["completed"] = missionCompleted;
  status["aborted"] = missionAborted;
  status["step"] = currentStep;
  status["total_steps"] = totalSteps;
  status["action"] = action;
  status["gps"] = gps_valid;
  status["gps_age_ms"] = last_gps_ms ? (millis() - last_gps_ms) : 4294967295UL;
  status["gps_raw_age_ms"] = last_gps_raw_ms ? (millis() - last_gps_raw_ms) : 4294967295UL;
  status["gps_fix_type"] = gps_fix_type;
  status["gps_satellites"] = gps_satellites;
  status["heartbeat_age_ms"] = last_heartbeat_ms ? (millis() - last_heartbeat_ms) : 4294967295UL;
  status["ekf_age_ms"] = last_ekf_ms ? (millis() - last_ekf_ms) : 4294967295UL;
  status["ekf_flags"] = ekf_flags;
  status["landed_state"] = drone_landed_state;
  status["lat"] = current_lat;
  status["lon"] = current_lon;
  status["alt"] = current_alt;
  status["armed"] = drone_armed;
  status["mode"] = drone_custom_mode;
  status["last_ack_command"] = last_ack_command;
  status["last_ack_result"] = last_ack_result;
  status["last_ack_age_ms"] = last_ack_ms ? (millis() - last_ack_ms) : 4294967295UL;
  status["last_abort"] = lastAbortReason;

  char out[1024];
  size_t n = serializeJson(status, out, sizeof(out));
  missionUdp.beginPacket(missionUdp.remoteIP(), missionUdp.remotePort());
  missionUdp.write((const uint8_t*)out, n);
  missionUdp.endPacket();
}

// ─── Telemetry Freshness ─────────────────────────────────────────────────────
bool telemetryFresh() {
  return gps_valid
      && last_gps_ms
      && last_gps_raw_ms
      && last_heartbeat_ms
      && (millis() - last_gps_ms <= TELEMETRY_FRESH_MS)
      && (millis() - last_gps_raw_ms <= TELEMETRY_FRESH_MS)
      && (millis() - last_heartbeat_ms <= TELEMETRY_FRESH_MS);
}

bool heartbeatFresh() {
  return last_heartbeat_ms && (millis() - last_heartbeat_ms <= TELEMETRY_FRESH_MS);
}

bool commandRejectedSince(uint16_t cmd, uint32_t since_ms) {
  return last_ack_ms >= since_ms
      && last_ack_command == cmd
      && last_ack_result != 0   // MAV_RESULT_ACCEPTED
      && last_ack_result != 5;  // MAV_RESULT_IN_PROGRESS
}

// ─── MAVLink TX ──────────────────────────────────────────────────────────────
// All outbound MAVLink 1 messages funnel through sendMavlink1() so the
// header/sequence/CRC framing lives in exactly one place.
static uint8_t tx_seq = 0;

void sendMavlink1(uint8_t msg_id, uint8_t crc_extra_val, const uint8_t* payload, uint8_t payload_len) {
  uint8_t pkt[6 + MAV_MAX_PAYLOAD + 2];
  pkt[0] = 0xFE;
  pkt[1] = payload_len;
  pkt[2] = tx_seq++;
  pkt[3] = MY_SYS_ID;
  pkt[4] = MY_COMP_ID;
  pkt[5] = msg_id;
  memcpy(&pkt[6], payload, payload_len);

  uint16_t crc = 0xFFFF;
  for (int i = 1; i < 6 + payload_len; i++) crc = crc_acc(pkt[i], crc);
  crc = crc_acc(crc_extra_val, crc);
  pkt[6 + payload_len]     = crc & 0xFF;
  pkt[6 + payload_len + 1] = (crc >> 8) & 0xFF;

  Serial2.write(pkt, 6 + payload_len + 2);
}

void sendCommandLong(uint16_t cmd,
                      float p1, float p2, float p3, float p4,
                      float p5, float p6, float p7) {
  uint8_t payload[33];
  float params[7] = {p1, p2, p3, p4, p5, p6, p7};
  memcpy(&payload[0], params, 28);
  memcpy(&payload[28], &cmd, 2);
  payload[30] = DRONE_SYS_ID;
  payload[31] = DRONE_COMP_ID;
  payload[32] = 0; // confirmation
  sendMavlink1(76, 152, payload, 33);
  Serial.printf("[TX] COMMAND_LONG cmd=%u\n", cmd);
}

void sendCommandInt(uint16_t cmd, uint8_t frame,
                     float p1, float p2, float p3, float p4,
                     int32_t x, int32_t y, float z) {
  uint8_t payload[35];
  memcpy(&payload[0],  &p1, 4);
  memcpy(&payload[4],  &p2, 4);
  memcpy(&payload[8],  &p3, 4);
  memcpy(&payload[12], &p4, 4);
  memcpy(&payload[16], &x, 4);
  memcpy(&payload[20], &y, 4);
  memcpy(&payload[24], &z, 4);
  memcpy(&payload[28], &cmd, 2);
  payload[30] = DRONE_SYS_ID;
  payload[31] = DRONE_COMP_ID;
  payload[32] = frame;
  payload[33] = 0; // current
  payload[34] = 0; // autocontinue
  sendMavlink1(75, 158, payload, 35);
  Serial.printf("[TX] COMMAND_INT cmd=%u frame=%u\n", cmd, frame);
}

// Shared builder for SET_POSITION_TARGET_LOCAL_NED (msg 84) and
// SET_POSITION_TARGET_GLOBAL_INT (msg 86) — identical wire layout, they only
// differ in how field1/field2/field3 and the frame byte are interpreted.
void sendPositionTarget(uint8_t msg_id, uint8_t frame, uint16_t type_mask,
                         const void* field1, const void* field2, const void* field3,
                         float vx, float vy, float vz) {
  uint8_t payload[53] = {0};
  uint32_t time_boot_ms = millis();
  memcpy(&payload[0],  &time_boot_ms, 4);
  memcpy(&payload[4],  field1, 4);
  memcpy(&payload[8],  field2, 4);
  memcpy(&payload[12], field3, 4);
  memcpy(&payload[16], &vx, 4);
  memcpy(&payload[20], &vy, 4);
  memcpy(&payload[24], &vz, 4);
  // bytes 28..47 (afx,afy,afz,yaw,yaw_rate) stay zero — ignored via type_mask
  memcpy(&payload[48], &type_mask, 2);
  payload[50] = DRONE_SYS_ID;
  payload[51] = DRONE_COMP_ID;
  payload[52] = frame;
  sendMavlink1(msg_id, msg_id == 86 ? 5 : 143, payload, 53);
}

void sendHeartbeat() {
  uint8_t payload[9] = {0};
  payload[4] = 6;   // type: MAV_TYPE_GCS
  payload[5] = 8;   // autopilot: MAV_AUTOPILOT_INVALID
  payload[6] = 192; // base_mode
  payload[7] = 0;   // system_status
  payload[8] = 3;   // mavlink_version
  sendMavlink1(0, 50, payload, 9);
}

void cmdArm(bool arm) {
  sendCommandLong(MAV_CMD_COMPONENT_ARM_DISARM, arm ? 1.0f : 0.0f, 0, 0, 0, 0, 0, 0);
}

void cmdTakeoff(float alt_m) {
  sendCommandLong(MAV_CMD_NAV_TAKEOFF, 0, 0, 0, 0, 0, 0, alt_m);
}

void cmdRTL() {
  sendCommandLong(MAV_CMD_NAV_RETURN_TO_LAUNCH, 0, 0, 0, 0, 0, 0, 0);
}

void cmdLand() {
  sendCommandLong(MAV_CMD_NAV_LAND, 0, 0, 0, 0, 0, 0, 0);
}

// Modern, officially-recommended way to change flight mode (MAV_CMD_DO_SET_MODE
// via COMMAND_LONG) instead of the deprecated MAVLink1 SET_MODE message.
void cmdSetMode(uint32_t custom_mode) {
  const float BASE_MODE_CUSTOM_ENABLED = 1.0f; // MAV_MODE_FLAG_CUSTOM_MODE_ENABLED
  sendCommandLong(MAV_CMD_DO_SET_MODE, BASE_MODE_CUSTOM_ENABLED, (float)custom_mode, 0, 0, 0, 0, 0);
  Serial.printf("[TX] DO_SET_MODE custom=%u\n", custom_mode);
}

void requestMessageInterval(uint32_t msg_id, float hz) {
  float interval_us = hz > 0.0f ? 1000000.0f / hz : -1.0f;
  sendCommandLong(MAV_CMD_REQUEST_MESSAGE, (float)msg_id, interval_us, 0, 0, 0, 0, 0);
}

void cmdReposition(float lat, float lon, float alt) {
  int32_t lat_int = (int32_t)(lat * 1e7f);
  int32_t lon_int = (int32_t)(lon * 1e7f);
  sendCommandInt(MAV_CMD_DO_REPOSITION, 6 /*GLOBAL_RELATIVE_ALT_INT*/, -1.0f, 0, WP_RADIUS_M, NAN, lat_int, lon_int, alt);
}

// Absolute guided position target (SET_POSITION_TARGET_GLOBAL_INT). Safe to
// resend unchanged — the frame is relative to a fixed reference (home), not
// to wherever the vehicle currently is, so repeated sends are idempotent.
void cmdGuidedPosition(float lat, float lon, float alt) {
  int32_t lat_int = (int32_t)(lat * 1e7f);
  int32_t lon_int = (int32_t)(lon * 1e7f);
  sendPositionTarget(86, 6 /*GLOBAL_RELATIVE_ALT_INT*/, POSMASK_POSITION_ONLY,
                      &lat_int, &lon_int, &alt, 0, 0, 0);
  Serial.printf("[TX] SET_POSITION_TARGET_GLOBAL_INT lat=%.6f lon=%.6f alt=%.1f\n", lat, lon, alt);
}

// Zero-velocity guided hold (SET_POSITION_TARGET_LOCAL_NED, velocity-only).
// Per ArduPilot docs, velocity targets must be resent at least once every
// GUID_TIMEOUT (default 3s) or the vehicle falls back to a plain hover.
void cmdLocalVelocityTarget(float vx, float vy, float vz) {
  float zero = 0.0f;
  sendPositionTarget(84, 1 /*MAV_FRAME_LOCAL_NED*/, POSMASK_VELOCITY_ONLY,
                      &zero, &zero, &zero, vx, vy, vz);
  Serial.printf("[TX] SET_POSITION_TARGET_LOCAL_NED vel vx=%.1f vy=%.1f vz=%.1f\n", vx, vy, vz);
}

uint32_t getModeValue(const String& mode_name) {
  String m = mode_name;
  m.toUpperCase();
  if (m == "GUIDED") return MODE_GUIDED;
  if (m == "LOITER") return MODE_LOITER;
  if (m == "RTL")    return MODE_RTL;
  if (m == "LAND")   return MODE_LAND;
  if (m == "AUTO")   return MODE_AUTO;
  if (m == "MANUAL") return MODE_MANUAL;
  return UINT32_MAX;
}

// ─── Geo Helpers ─────────────────────────────────────────────────────────────
float haversine(float lat1, float lon1, float lat2, float lon2) {
  const float R = 6371000.0f;
  float dlat = radians(lat2 - lat1);
  float dlon = radians(lon2 - lon1);
  float a = sinf(dlat/2)*sinf(dlat/2)
           + cosf(radians(lat1))*cosf(radians(lat2))*sinf(dlon/2)*sinf(dlon/2);
  return R * 2.0f * atan2f(sqrtf(a), sqrtf(1.0f-a));
}

// Converts a North/East offset in meters (relative to wherever the vehicle
// is right now) into an absolute lat/lon, using an equirectangular
// approximation — accurate enough for the few-meter offsets these missions
// use. Computing this ONCE and sending the resulting absolute target is what
// makes "local_ned" safe to resend (see stepLocalNed below).
void offsetToLatLon(float lat, float lon, float north_m, float east_m, float* out_lat, float* out_lon) {
  const float EARTH_RADIUS_M = 6371000.0f;
  const float RAD2DEG = 180.0f / PI;
  *out_lat = lat + (north_m / EARTH_RADIUS_M) * RAD2DEG;
  *out_lon = lon + (east_m / (EARTH_RADIUS_M * cosf(radians(lat)))) * RAD2DEG;
}

// ─── Mission Execution ────────────────────────────────────────────────────────
void abortMission(const char* reason) {
  missionRunning = false;
  missionCompleted = false;
  missionAborted = true;
  strncpy(lastAbortReason, reason, sizeof(lastAbortReason) - 1);
  lastAbortReason[sizeof(lastAbortReason) - 1] = '\0';
  digitalWrite(RELAY_PIN, LOW);
  Serial.printf("[ABORT] %s — commanding LAND\n", reason);
  cmdLand();
}

void stepRelay(bool on) {
  digitalWrite(RELAY_PIN, on ? HIGH : LOW);
  Serial.printf("[Step] relay %s\n", on ? "ON" : "OFF");
  currentStep++;
}

void stepSetMode(JsonObject step) {
  if (!step["mode"].is<const char*>() && !step["mode"].is<String>()) {
    Serial.println("[Step] set_mode missing mode value — skipping");
    stepStartTime = 0;
    currentStep++;
    return;
  }
  String mode_name = step["mode"].as<String>();
  uint32_t target_mode = getModeValue(mode_name);
  if (target_mode == UINT32_MAX) {
    Serial.printf("[Step] unsupported mode '%s' — skipping\n", mode_name.c_str());
    stepStartTime = 0;
    currentStep++;
    return;
  }

  if (stepStartTime == 0) {
    cmdSetMode(target_mode);
    stepStartTime = millis();
    Serial.printf("[Step] requesting mode %s (custom=%u)\n", mode_name.c_str(), (unsigned int)target_mode);
  }

  if (drone_custom_mode == target_mode) {
    Serial.printf("[Step] mode %s confirmed\n", mode_name.c_str());
    stepStartTime = 0;
    currentStep++;
  } else if (millis() - stepStartTime > MODE_CONFIRM_TIMEOUT_MS) {
    Serial.printf("[Step] mode %s timeout — continuing anyway\n", mode_name.c_str());
    stepStartTime = 0;
    currentStep++;
  }
}

void stepWait(JsonObject step) {
  uint32_t dur_ms = (uint32_t)step["seconds"].as<int>() * 1000UL;
  if (stepStartTime == 0) {
    stepStartTime = millis();
    Serial.printf("[Step] waiting %u s\n", dur_ms / 1000);
  }
  if (millis() - stepStartTime >= dur_ms) {
    stepStartTime = 0;
    currentStep++;
  }
}

void stepArm() {
  if (drone_armed) {
    Serial.println("[Step] Already armed — skipping");
    stepStartTime = lastWpSendTime = 0;
    currentStep++;
    return;
  }
  if (stepStartTime == 0) {
    cmdSetMode(MODE_GUIDED);
    cmdArm(true);
    stepStartTime = lastWpSendTime = millis();
    Serial.println("[Step] ARM commanded — waiting for confirmation...");
  }
  if (millis() - lastWpSendTime >= ARM_RETRY_INTERVAL) {
    Serial.println("[Step] Re-sending ARM...");
    cmdSetMode(MODE_GUIDED);
    cmdArm(true);
    lastWpSendTime = millis();
  }
  if (commandRejectedSince(MAV_CMD_COMPONENT_ARM_DISARM, stepStartTime)) {
    stepStartTime = lastWpSendTime = 0;
    abortMission("ARM rejected by autopilot");
    return;
  }
  if (drone_armed) {
    Serial.println("[Step] Drone ARMED confirmed");
    stepStartTime = lastWpSendTime = 0;
    currentStep++;
  } else if (millis() - stepStartTime > ARM_TIMEOUT_MS) {
    stepStartTime = lastWpSendTime = 0;
    abortMission("ARM timeout or arming checks failed");
  }
}

void stepDisarm() {
  if (stepStartTime == 0) {
    cmdArm(false);
    stepStartTime = millis();
    Serial.println("[Step] DISARM commanded — waiting for confirmation...");
  }
  if (!drone_armed) {
    Serial.println("[Step] Drone DISARMED confirmed");
    stepStartTime = 0;
    currentStep++;
  } else if (millis() - stepStartTime > DISARM_TIMEOUT_MS) {
    Serial.println("[Step] DISARM timeout");
    stepStartTime = 0;
    currentStep++;
  } else if (commandRejectedSince(MAV_CMD_COMPONENT_ARM_DISARM, stepStartTime)) {
    Serial.println("[Step] DISARM rejected by autopilot");
    stepStartTime = 0;
    currentStep++;
  }
}

void stepTakeoff(JsonObject step) {
  float alt = step["alt"].as<float>();
  if (stepStartTime == 0) {
    cmdSetMode(MODE_GUIDED);
    cmdArm(true);
    stepStartTime = millis();
    lastWpSendTime = 0;
    Serial.printf("[Step] preparing takeoff to %.1fm\n", alt);
  }
  if (!drone_armed) {
    if (millis() - lastWpSendTime >= ARM_RETRY_INTERVAL) {
      cmdArm(true);
      lastWpSendTime = millis();
    }
    if (commandRejectedSince(MAV_CMD_COMPONENT_ARM_DISARM, stepStartTime)) {
      abortMission("takeoff arming rejected by autopilot");
    }
    if (millis() - stepStartTime > ARM_TIMEOUT_MS) abortMission("takeoff arm timeout");
    return;
  }
  if (lastWpSendTime == 0) {
    cmdTakeoff(alt);
    stepStartTime = lastWpSendTime = millis();
    Serial.printf("[Step] takeoff command sent to %.1fm\n", alt);
  }
  if (commandRejectedSince(MAV_CMD_NAV_TAKEOFF, stepStartTime)) {
    abortMission("takeoff rejected by autopilot");
    return;
  }
  if (telemetryFresh() && fabsf(current_alt - alt) <= ALT_REACHED_TOLERANCE_M) {
    Serial.println("[Step] takeoff altitude reached");
    stepStartTime = lastWpSendTime = 0;
    currentStep++;
  } else if (millis() - stepStartTime > TAKEOFF_TIMEOUT_MS) {
    abortMission("takeoff timeout");
  }
}

void stepReposition(JsonObject step) {
  float tlat = step["lat"].as<float>();
  float tlon = step["lon"].as<float>();
  float talt = step["alt"].as<float>();
  if (!telemetryFresh()) {
    if (stepStartTime == 0) stepStartTime = millis();
    if (millis() - stepStartTime > GPS_LOCK_TIMEOUT) abortMission("no fresh GPS/heartbeat telemetry");
    return;
  }
  if (stepStartTime == 0) {
    cmdSetMode(MODE_GUIDED);
    cmdReposition(tlat, tlon, talt);
    stepStartTime = lastWpSendTime = millis();
    Serial.printf("[Step] reposition lat=%.6f lon=%.6f alt=%.1f\n", tlat, tlon, talt);
  }
  if (millis() - lastWpSendTime >= WP_SEND_INTERVAL) {
    cmdReposition(tlat, tlon, talt);
    lastWpSendTime = millis();
  }
  if (commandRejectedSince(MAV_CMD_DO_REPOSITION, stepStartTime)) {
    abortMission("reposition rejected by autopilot");
    return;
  }
  if (millis() - stepStartTime > WP_TIMEOUT_MS) { abortMission("reposition timeout"); return; }

  float dist = haversine(current_lat, current_lon, tlat, tlon);
  float alt_err = fabsf(current_alt - talt);
  Serial.printf("[Reposition] dist=%.1fm alt_err=%.1fm\n", dist, alt_err);
  if (dist <= WP_RADIUS_M && alt_err <= WP_ALT_ERR_M) {
    Serial.println("[Step] reposition target REACHED");
    stepStartTime = lastWpSendTime = 0;
    currentStep++;
  }
}

void stepFlyTo(JsonObject step) {
  float tlat = step["lat"].as<float>();
  float tlon = step["lon"].as<float>();
  float talt = step["alt"].as<float>();
  if (!telemetryFresh()) {
    if (stepStartTime == 0) stepStartTime = millis();
    if (millis() - stepStartTime > GPS_LOCK_TIMEOUT) abortMission("no fresh GPS/heartbeat telemetry");
    return;
  }
  if (stepStartTime == 0) {
    cmdSetMode(MODE_GUIDED);
    cmdGuidedPosition(tlat, tlon, talt);
    stepStartTime = lastWpSendTime = millis();
    Serial.printf("[Step] fly_to lat=%.6f lon=%.6f alt=%.1f\n", tlat, tlon, talt);
  }
  if (millis() - lastWpSendTime >= WP_SEND_INTERVAL) {
    cmdGuidedPosition(tlat, tlon, talt);
    lastWpSendTime = millis();
  }
  if (millis() - stepStartTime > WP_TIMEOUT_MS) { abortMission("waypoint timeout"); return; }

  float dist = haversine(current_lat, current_lon, tlat, tlon);
  float alt_err = fabsf(current_alt - talt);
  Serial.printf("[Nav] dist=%.1fm alt_err=%.1fm\n", dist, alt_err);
  if (dist <= WP_RADIUS_M && alt_err <= WP_ALT_ERR_M) {
    Serial.println("[Step] waypoint REACHED");
    stepStartTime = lastWpSendTime = 0;
    currentStep++;
  }
}

// Moves N meters north / E meters east / D meters down of wherever the
// vehicle is when the step starts, then holds there for "seconds".
//
// IMPORTANT: the target lat/lon is computed ONCE at step start and then
// resent as a fixed ABSOLUTE position (via cmdGuidedPosition), never as a
// MAV_FRAME_LOCAL_OFFSET_NED offset. That frame means "relative to the
// vehicle's position right now" for every message received — resending the
// same offset on the periodic refresh timer previously caused the vehicle
// to be commanded further in the same direction on every resend, producing
// a runaway drift. Converting to a fixed absolute target once removes that
// failure mode entirely and makes local_ned behave like fly_to/reposition.
void stepLocalNed(JsonObject step) {
  float north_m = step["x"].as<float>();
  float east_m  = step["y"].as<float>();
  float down_m  = step["z"].as<float>();
  uint32_t hold_ms = (uint32_t)(step["seconds"] | 0) * 1000UL;

  if (!telemetryFresh()) {
    if (stepStartTime == 0) stepStartTime = millis();
    if (millis() - stepStartTime > GPS_LOCK_TIMEOUT) abortMission("no fresh GPS/heartbeat telemetry");
    return;
  }

  if (stepStartTime == 0) {
    offsetToLatLon(current_lat, current_lon, north_m, east_m, &stepTargetLat, &stepTargetLon);
    stepTargetAlt = current_alt - down_m;
    cmdSetMode(MODE_GUIDED);
    cmdGuidedPosition(stepTargetLat, stepTargetLon, stepTargetAlt);
    stepStartTime = lastWpSendTime = millis();
    stepHoldStartTime = 0;
    Serial.printf("[Step] local_ned north=%.1fm east=%.1fm down=%.1fm -> lat=%.6f lon=%.6f alt=%.1f\n",
                  north_m, east_m, down_m, stepTargetLat, stepTargetLon, stepTargetAlt);
  }
  if (millis() - lastWpSendTime >= WP_SEND_INTERVAL) {
    cmdGuidedPosition(stepTargetLat, stepTargetLon, stepTargetAlt);
    lastWpSendTime = millis();
  }
  if (millis() - stepStartTime > WP_TIMEOUT_MS) { abortMission("local_ned timeout"); return; }

  float dist = haversine(current_lat, current_lon, stepTargetLat, stepTargetLon);
  float alt_err = fabsf(current_alt - stepTargetAlt);
  bool arrived = (dist <= WP_RADIUS_M && alt_err <= WP_ALT_ERR_M);

  if (arrived && stepHoldStartTime == 0) {
    stepHoldStartTime = millis();
    Serial.println("[Step] local_ned target reached — holding");
  }
  if (arrived && millis() - stepHoldStartTime >= hold_ms) {
    stepStartTime = lastWpSendTime = stepHoldStartTime = 0;
    currentStep++;
  }
}

void stepBrakeHover(JsonObject step) {
  uint32_t hold_ms = (uint32_t)(step["seconds"] | 5) * 1000UL;
  if (stepStartTime == 0) {
    cmdSetMode(MODE_GUIDED);
    cmdLocalVelocityTarget(0, 0, 0);
    stepStartTime = lastWpSendTime = millis();
    Serial.printf("[Step] brake_hover zero velocity for %us\n", hold_ms / 1000);
  }
  if (millis() - lastWpSendTime >= VELOCITY_RESEND_INTERVAL) {
    cmdLocalVelocityTarget(0, 0, 0);
    lastWpSendTime = millis();
  }
  if (millis() - stepStartTime >= hold_ms) {
    stepStartTime = lastWpSendTime = 0;
    currentStep++;
  }
}

void stepRtl() {
  cmdRTL();
  Serial.println("[Step] RTL commanded");
  currentStep++;
}

void stepLand() {
  if (stepStartTime == 0) {
    cmdLand();
    stepStartTime = lastWpSendTime = millis();
    Serial.println("[Step] LAND commanded — waiting for landed confirmation...");
  }
  if (millis() - lastWpSendTime >= LAND_RESEND_INTERVAL) {
    cmdLand();
    lastWpSendTime = millis();
  }
  if (commandRejectedSince(MAV_CMD_NAV_LAND, stepStartTime)) {
    abortMission("LAND rejected by autopilot");
    return;
  }
  if (drone_landed_state == MAV_LANDED_STATE_ON_GROUND ||
      (!drone_armed && millis() - stepStartTime > LAND_DISARM_GRACE_MS)) {
    Serial.println("[Step] LAND confirmed");
    stepStartTime = lastWpSendTime = 0;
    currentStep++;
  } else if (millis() - stepStartTime > LAND_CONFIRM_TIMEOUT_MS) {
    abortMission("LAND confirmation timeout");
  }
}

void executeMission() {
  JsonArray steps = missionDoc["steps"];
  if (!steps || currentStep >= (int)steps.size()) {
    missionRunning = false;
    missionCompleted = true;
    missionAborted = false;
    digitalWrite(RELAY_PIN, LOW);
    Serial.println("[Mission] COMPLETE");
    return;
  }

  JsonObject step = steps[currentStep];
  String action = step["action"].as<String>();

  if      (action == "relay_on")    stepRelay(true);
  else if (action == "relay_off")   stepRelay(false);
  else if (action == "set_mode")    stepSetMode(step);
  else if (action == "wait")        stepWait(step);
  else if (action == "arm")         stepArm();
  else if (action == "disarm")      stepDisarm();
  else if (action == "takeoff")     stepTakeoff(step);
  else if (action == "reposition")  stepReposition(step);
  else if (action == "fly_to")      stepFlyTo(step);
  else if (action == "local_ned")   stepLocalNed(step);
  else if (action == "brake_hover") stepBrakeHover(step);
  else if (action == "rtl")         stepRtl();
  else if (action == "land")        stepLand();
  else {
    Serial.printf("[Step] unknown action '%s' — skipping\n", action.c_str());
    currentStep++;
  }
}

// ─── Mission UDP Commands ─────────────────────────────────────────────────────
void handleMissionCommand(const String& msg) {
  if (msg == "START") {
    if (missionRunning) {
      Serial.println("[UDP] WARNING: already running — ignoring START");
      sendUdpText("ERROR already_running");
    } else if (!missionDoc["steps"].is<JsonArray>()) {
      Serial.println("[UDP] ERROR: no mission stored yet");
      sendUdpText("ERROR no_mission");
    } else if (!heartbeatFresh()) {
      Serial.println("[UDP] ERROR: no fresh Pixhawk heartbeat yet");
      sendUdpText("ERROR no_fresh_heartbeat");
    } else {
      currentStep = 0; stepStartTime = 0; lastWpSendTime = 0; stepHoldStartTime = 0;
      missionCompleted = false;
      missionAborted = false;
      lastAbortReason[0] = '\0';
      missionRunning = true;
      Serial.println("[UDP] Mission START");
      sendUdpText("OK START");
    }
  } else if (msg == "STOP") {
    abortMission("STOP command received");
    sendUdpText("OK STOP");
  } else if (msg == "STATUS") {
    Serial.printf("[STATUS] step=%d running=%d gps=%d lat=%.6f lon=%.6f alt=%.2f armed=%d\n",
                  currentStep, missionRunning, gps_valid,
                  current_lat, current_lon, current_alt, drone_armed);
    sendStatusReply();
  } else if (msg.startsWith("{")) {
    DeserializationError err = deserializeJson(missionDoc, msg);
    Serial.println(err ? "[UDP] JSON parse FAILED" : "[UDP] Mission stored OK");
    if (err) {
      sendUdpText("ERROR json_parse_failed");
    } else {
      currentStep = 0; stepStartTime = 0; lastWpSendTime = 0; stepHoldStartTime = 0;
      missionCompleted = false;
      missionAborted = false;
      lastAbortReason[0] = '\0';
      sendUdpText("OK MISSION_STORED");
    }
  }
}

void handleMissionUdp() {
  int packetSize = missionUdp.parsePacket();
  if (!packetSize) return;

  char buf[2048];
  int len = missionUdp.read(buf, sizeof(buf) - 1);
  if (len <= 0) return;
  buf[len] = '\0';

  String msg = String(buf);
  msg.trim();
  handleMissionCommand(msg);
}

// ─── Setup / Loop ─────────────────────────────────────────────────────────────
uint32_t lastHeartbeatTime = 0;
uint32_t lastTelemetryRequestTime = 0;

void setup() {
  Serial.begin(115200);
  Serial2.begin(SERIAL_BAUD, SERIAL_8N1, RXD2, TXD2);

  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW);

  WiFi.softAP(WIFI_SSID, WIFI_PASS);
  mavUdp.begin(MAVLINK_UDP_PORT);
  missionUdp.begin(MISSION_UDP_PORT);

  Serial.println("[ESP32] Ready");
  Serial.printf("[ESP32] AP IP: %s  MAVLink UDP: %d  Mission UDP: %d\n",
                WiFi.softAPIP().toString().c_str(), MAVLINK_UDP_PORT, MISSION_UDP_PORT);
}

void loop() {
  if (millis() - lastHeartbeatTime >= 1000) {
    sendHeartbeat();
    lastHeartbeatTime = millis();
  }

  if (millis() - lastTelemetryRequestTime >= 5000) {
    requestMessageInterval(MAVLINK_MSG_GPS_RAW_INT, 2.0f);
    requestMessageInterval(MAVLINK_MSG_GLOBAL_POSITION_INT, 5.0f);
    requestMessageInterval(MAVLINK_MSG_EKF_STATUS_REPORT, 2.0f);
    requestMessageInterval(MAVLINK_MSG_EXTENDED_SYS_STATE, 2.0f);
    lastTelemetryRequestTime = millis();
  }

  parsePixhawkTelemetry();
  handleMavlinkUdp();
  handleMissionUdp();

  if (missionRunning) executeMission();
}
