#include <WiFi.h>
#include <WiFiAP.h>
#include <WiFiUdp.h>
#include <MAVLink.h>
#include <HardwareSerial.h>

// ======================================================
// PIXHAWK UART
// ======================================================
#define RXD2 16
#define TXD2 17

HardwareSerial pxSerial(2);

// ======================================================
// RELAY PIN
// ======================================================
#define RELAY_PIN 4

// ======================================================
// WIFI
// ======================================================
const char* ssid = "MAVLink";
const char* password = "12345678";

// ======================================================
// UDP
// ======================================================
WiFiUDP udp;
IPAddress broadcastIP(255, 255, 255, 255);
const int udpPort = 14550;

void setup() {

  Serial.begin(115200);

  // Pixhawk UART
  pxSerial.begin(57600, SERIAL_8N1, RXD2, TXD2);

  // Relay setup
  pinMode(RELAY_PIN, OUTPUT);

  // SAFE DEFAULT STATE (RELAY OFF)
  digitalWrite(RELAY_PIN, LOW);

  // Start WiFi AP
  WiFi.softAP(ssid, password);

  Serial.println();
  Serial.println("=================================");
  Serial.println("ESP32 MAVLink + Relay Bridge");
  Serial.println("=================================");

  Serial.print("ESP32 IP: ");
  Serial.println(WiFi.softAPIP());

  // Start UDP
  udp.begin(udpPort);

  Serial.println("READY");
}

void loop() {

  // ======================================================
  // PIXHAWK -> WIFI
  // ======================================================
  while (pxSerial.available()) {

    uint8_t c = pxSerial.read();

    udp.beginPacket(broadcastIP, udpPort);
    udp.write(&c, 1);
    udp.endPacket();
  }

  // ======================================================
  // WIFI -> ESP32
  // ======================================================
  int packetSize = udp.parsePacket();

  if (packetSize) {

    char incoming[255];
    int len = udp.read(incoming, 255);

    if (len > 0) {

      incoming[len] = '\0';
      String msg = String(incoming);

      Serial.print("Received: ");
      Serial.println(msg);

      // ==================================================
      // RELAY ON (FIXED LOGIC)
      // ==================================================
      if (msg == "RELAY_ON") {

        Serial.println("Relay ON");
        digitalWrite(RELAY_PIN, HIGH);
      }

      // ==================================================
      // RELAY OFF (FIXED LOGIC)
      // ==================================================
      if (msg == "RELAY_OFF") {

        Serial.println("Relay OFF");
        digitalWrite(RELAY_PIN, LOW);
      }

      // Forward MAVLink packets to Pixhawk
      pxSerial.write((uint8_t*)incoming, len);
    }
  }
}
